let isSyncingToCloud = false; async function syncDataToCloud() { // Chặn đứng các yêu cầu đồng bộ dồn dập (double-tap/spam) từ điện thoại hoặc mạng trễ if (isSyncingToCloud) return; isSyncingToCloud = true; // Lọc chống trùng lặp bệnh nhân theo ID trước khi lưu const uniquePatientsMap = {}; patients.forEach(p => { if (p && p.id) uniquePatientsMap[String(p.id)] = p; }); patients = Object.values(uniquePatientsMap); localStorage.setItem('bs_patients_scroll_v3', JSON.stringify(patients)); localStorage.setItem('bs_room_configs_v21', JSON.stringify(roomList)); localStorage.setItem('bs_med_categories_v14', JSON.stringify(medCategories)); renderDashboardCounts(); if (!GOOGLE_SCRIPT_URL) { isSyncingToCloud = false; return; } updateSyncUI('syncing', 'Đang lưu...'); try { const payload = JSON.stringify({ patients, rooms: roomList, medCategories }); if (navigator.sendBeacon) { const blob = new Blob([payload], { type: 'text/plain; charset=UTF-8' }); navigator.sendBeacon(GOOGLE_SCRIPT_URL, blob); } else { await fetch(GOOGLE_SCRIPT_URL, { method: 'POST', mode: 'no-cors', cache: 'no-cache', headers: { 'Content-Type': 'text/plain;charset=utf-8' }, body: payload }); } updateSyncUI('success', 'Đã lưu'); } catch { updateSyncUI('error', 'Không thể lưu'); } finally { // Mở khóa an toàn sau 1.5 giây để các thao tác nhập/xóa diễn ra ổn định, tuần tự setTimeout(() => { isSyncingToCloud = false; }, 1500); } } function updateSyncUI(st, txt) { const el = document.getElementById('syncStatus'); if (!el) return; el.textContent = (st === 'syncing' ? '🔄 ' : st === 'success' ? '☁️ ' : '⚠️ ') + txt; el.style.background = st === 'syncing' ? '#fef3c7' : st === 'success' ? '#e0f2fe' : '#fee2e2'; el.style.color = st === 'syncing' ? '#92400e' : st === 'success' ? '#0369a1' : '#991b1b'; // Tránh việc giao diện bị kẹt chữ "Đang..." khi mạng bị chập chờn if (st === 'syncing') { setTimeout(() => { if (el.textContent.includes('Đang')) { el.textContent = '☁️ Đã đồng bộ'; el.style.background = '#e0f2fe'; el.style.color = '#0369a1'; } }, 4000); } }
