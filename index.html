async function loadDataFromCloud() { if (!GOOGLE_SCRIPT_URL) return; updateSyncUI('syncing', 'Đang nạp...'); try { const res = await fetch(GOOGLE_SCRIPT_URL); const data = await res.json(); if (data?.patients) { patients = data.patients; localStorage.setItem('bs_patients_scroll_v3', JSON.stringify(patients)); } if (data?.medCategories && data.medCategories.length > 0) { medCategories = data.medCategories; localStorage.setItem('bs_med_categories_v14', JSON.stringify(medCategories)); } updateSyncUI('success', 'Đã đồng bộ'); renderDashboardCounts(); renderMedCategoriesUI(); } catch { updateSyncUI('error', 'Lỗi kết nối'); } } async function syncDataToCloud() { localStorage.setItem('bs_patients_scroll_v3', JSON.stringify(patients)); localStorage.setItem('bs_room_configs_v21', JSON.stringify(roomList)); localStorage.setItem('bs_med_categories_v14', JSON.stringify(medCategories)); renderDashboardCounts(); if (!GOOGLE_SCRIPT_URL) return; updateSyncUI('syncing', 'Đang lưu...'); try { await fetch(GOOGLE_SCRIPT_URL, { method: 'POST', mode: 'no-cors', headers: {'Content-Type': 'text/plain'}, body: JSON.stringify({ patients, rooms: roomList, medCategories }) }); updateSyncUI('success', 'Đã lưu'); } catch { updateSyncUI('error', 'Không thể lưu'); } }
